All previous items are rendered as links. {% for item in items %} {% set is_last = loop.last %} {% set is_linked = item.href %} {% if is_linked %} {% snippet 'v2/components/text-link.html', label=item.label, href=item.href, style='tertiary', size='xs', attrs={'property': 'item', 'typeof': 'WebPage'}, inner_attrs={'property': 'name'} %} {% else %} {{ item.label }} {% endif %} {% if not is_last %} {{ separator }} {% endif %} {% endfor %}
