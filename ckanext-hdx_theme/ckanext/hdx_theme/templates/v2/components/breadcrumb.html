{# Snippet: breadcrumb.html Path: templates/v2/components/breadcrumb.html Renders a c-breadcrumb horizontal trail. Parameters ───────────────────────────────────────────────── items {list} Ordered list of dicts: {label, href}. The LAST item is rendered as the current (non-linked) crumb. All previous items are rendered as links. default: [] show_home {bool} When true (default), prepend a Home link as the first crumb. Set to false on pages that manage the first crumb themselves. default: true separator {string} Separator character between crumbs. default: '/' extra_classes {string} default: '' ───────────────────────────────────────────────── Example (Home prepended automatically) {% snippet 'v2/components/breadcrumb.html', items=[ {'label': 'Sudan', 'href': '/dataset?location=sd'}, {'label': 'Humanitarian data', 'href': ''}, ] %} Example (Home omitted) {% snippet 'v2/components/breadcrumb.html', show_home=false, items=[ {'label': 'Datasets', 'href': '/dataset'}, {'label': 'My title', 'href': ''}, ] %} #} {% set items = items if items is defined else [] %} {% set show_home = show_home if show_home is defined else true %} {% set separator = separator if separator is defined else '/' %} {% set extra_classes = extra_classes if extra_classes is defined else '' %} {% if show_home %} {% set items = [{'label': _('Home'), 'href': '/'}] + items %} {% endif %}
{% for item in items %} {% set is_last = loop.last %} {% set is_linked = item.href %} {% if is_linked %} {% snippet 'v2/components/text-link.html', label=item.label, href=item.href, style='tertiary', size='xs', attrs={'property': 'item', 'typeof': 'WebPage'}, inner_attrs={'property': 'name'} %} {% else %} {{ item.label }} {% endif %} {% if not is_last %} {{ separator }} {% endif %} {% endfor %}
